export function formatIndianCurrency(amount: number): string {
  if (isNaN(amount)) return '₹0.00';
  return new Intl.NumberFormat('en-IN', {
    style: 'currency',
    currency: 'INR',
    minimumFractionDigits: 2,
    maximumFractionDigits: 2,
  }).format(amount);
}

Utility 2: UPI SMS Transaction Parser
--------------------------------------
export interface ParsedTransaction {
  amount: number;
  type: 'DEBIT' | 'CREDIT';
  merchant: string;
  referenceId?: string;
  rawSms: string;
}

export function parseUPISms(smsBody: string): ParsedTransaction | null {
  if (!smsBody || typeof smsBody !== 'string') return null;

  const amountRegex = /(?:Rs\.?|INR|₹)\s*([0-9,]+\.?[0-9]*)/i;
  const amountMatch = smsBody.match(amountRegex);
  if (!amountMatch) return null;

  const amount = parseFloat(amountMatch[1].replace(/,/g, ''));
  if (isNaN(amount) || amount <= 0) return null;

  const isCredit = /(?:credited|received|added|deposited|cashback|refund)/i.test(smsBody);
  const type: 'DEBIT' | 'CREDIT' = isCredit ? 'CREDIT' : 'DEBIT';

  let merchant = 'Unknown Merchant';
  const merchantMatch = smsBody.match(/(?:to|at|from|VPA)\s+([A-Za-z0-9@\.\-_]+(?:\s+[A-Za-z0-9@\.\-_]+)?)/i);
  if (merchantMatch && merchantMatch[1]) {
    merchant = merchantMatch[1].trim().replace(/\b(on|ref|upi|via|A\/c)\b/gi, '').trim();
  }

  const refMatch = smsBody.match(/(?:Ref|UPI Ref|UTR|IMPS|Txn)(?:\s*No\.?)?[:\s]*([0-9a-zA-Z]+)/i);
  const referenceId = refMatch ? refMatch[1] : undefined;

  return { amount, type, merchant, referenceId, rawSms: smsBody };
}
